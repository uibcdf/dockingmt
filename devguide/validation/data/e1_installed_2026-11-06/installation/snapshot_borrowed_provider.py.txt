EXPECTED_NATIVE = '2a79b9a83b0fe0b00a54c9b80f04bf6c9ace2b3473c53269edbfe29904f8930b'
EXCLUDE = {'__pycache__', '.git', '.pytest_cache', 'build', 'dist', 'target', 'docs', 'devguide'}


def sha(path):
    with path.open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()


def git(*args):
    return subprocess.check_output(['git', '-C', str(SOURCE), *args], text=True).strip()


def available():
    return int(next(line.split()[1] for line in Path('/proc/meminfo').read_text().splitlines()
                    if line.startswith('MemAvailable:'))) * 1024


def headroom():
    if available() < 8 * 1024**3:
        raise RuntimeError('Available-RAM reserve threatened; no automatic retry')
    if resource.getrusage(resource.RUSAGE_SELF).ru_maxrss * 1024 > 1024**3:
        raise RuntimeError('Task RSS limit exceeded; no automatic retry')


def copy_tree(origin, relative, role):
    allowed = origin.resolve()
    for path in sorted(origin.rglob('*')):
        rel = path.relative_to(origin)
        if any(part in EXCLUDE for part in rel.parts) or path.suffix in {'.pyc', '.pyo'}:
            continue
        if path.is_symlink() and path.is_dir():
            raise RuntimeError('Directory symlink requires explicit review: ' + str(path))
        if path.is_dir():
            continue
        resolved = path.resolve()
        if not resolved.is_relative_to(allowed) or not resolved.is_file():
            raise RuntimeError('Source file escapes its declared runtime root: ' + str(path))
        headroom()
        destination = TARGET / relative / rel
        if destination.exists():
            raise RuntimeError('Refusing to overwrite a target file: ' + str(destination))
        destination.parent.mkdir(parents=True, exist_ok=True)
        digest = sha(resolved)
        shutil.copy2(resolved, destination)
        if sha(destination) != digest or sha(resolved) != digest:
            raise RuntimeError('Copy/source changed during snapshot: ' + str(path))
        records.append(dict(role=role, relative_path=str(destination.relative_to(TARGET)),
                            source_path=str(path), source_resolved=str(resolved),
                            target_path=str(destination), sha256=digest,
                            bytes=destination.stat().st_size,
                            source_was_symlink=path.is_symlink()))
