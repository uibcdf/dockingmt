def sha(path):
    with path.open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()


def write(path, value):
    with path.open('x') as stream:
        json.dump(value, stream, indent=2, allow_nan=False)
        stream.write('\n')


def available():
    return int(next(line.split()[1] for line in Path('/proc/meminfo').read_text().splitlines()
                    if line.startswith('MemAvailable:'))) * 1024


def git(path, *args):
    return subprocess.check_output(['git', '-C', str(path), *args], text=True).strip()


def notebook_info(path):
    nb = json.loads(path.read_text())
    rows = []
    budgets = None
    for cell in nb['cells']:
        if cell['cell_type'] != 'code':
            continue
        body = ''.join(cell['source'])
        tree = ast.parse(body)
        assert cell['execution_count'] is None and cell['outputs'] == []
        for node in tree.body:
            if isinstance(node, ast.Assign) and any(isinstance(t, ast.Name) and t.id == 'budgets' for t in node.targets):
                budgets = {kw.arg: ast.literal_eval(kw.value) for kw in node.value.keywords}
        rows.append(dict(id=cell['id'], sha256=hashlib.sha256(body.encode()).hexdigest()))
    return dict(path=str(path.relative_to(root)), sha256=sha(path), code_cells=rows,
                monitored_direct_budgets=budgets)


def group_rss(group):
    total, owned = 0, 0
    for directory in Path('/proc').iterdir():
        if not directory.name.isdigit():
            continue
        try:
            tail = (directory / 'stat').read_text().rsplit(') ', 1)[1].split()
            if int(tail[2]) != group:
                continue
            owned += 1
            fields = (directory / 'status').read_text().splitlines()
            entry = next((line for line in fields if line.startswith('VmRSS:')), None)
            if entry is not None:
                total += int(entry.split()[1]) * 1024
        except (FileNotFoundError, ProcessLookupError, PermissionError):
            continue
    return total, owned


def limits():
    resource.setrlimit(resource.RLIMIT_AS, (address_limit, address_limit))
    resource.setrlimit(resource.RLIMIT_CPU, (60, 60))


def run(phase, extra):
    assert available() >= reserve + rss_limit
    env = os.environ.copy()
    for key in ['OMP_NUM_THREADS', 'OMP_THREAD_LIMIT', 'OPENBLAS_NUM_THREADS',
                'MKL_NUM_THREADS', 'NUMEXPR_NUM_THREADS', 'BLIS_NUM_THREADS',
                'RAYON_NUM_THREADS', 'CARGO_BUILD_JOBS', 'MAX_JOBS']:
        env[key] = '1'
    env.update(PYTHONDONTWRITEBYTECODE='1', XDG_CACHE_HOME=str(attempt / (phase + '-cache')),
               DOCKINGMT_E1_ROOT=str(root), DOCKINGMT_E1_INSTALLED_TARGET=str(task / 'site-packages'),
               DOCKINGMT_E1_PREPARATION_AMENDMENT=str(task / 'preparation-correction.json'),
               LD_LIBRARY_PATH=str(routine / 'lib'))
    env.update(extra)
    started = time.monotonic()
    peak, minimum, samples, max_owned, interruption = 0, available(), 0, 0, None
    with (attempt / (phase + '-stdout.log')).open('xb') as stdout, (attempt / (phase + '-stderr.log')).open('xb') as stderr:
        process = subprocess.Popen([str(python), '-B', '-I', '-S', str(attempt / 'worker.py'), str(attempt), phase],
                                   cwd=attempt, env=env, stdout=stdout, stderr=stderr,
                                   preexec_fn=limits, start_new_session=True)
        while process.poll() is None:
            current = available()
            minimum = min(minimum, current)
            resident, owned = group_rss(process.pid)
            peak, max_owned = max(peak, resident), max(max_owned, owned)
            samples += 1
            if current < reserve:
                interruption = 'available_RAM_reserve_threatened'
            elif resident > rss_limit:
                interruption = 'owned_process_group_RSS_limit'
            elif time.monotonic() - started > 90:
                interruption = 'wall_limit'
            if interruption:
                os.killpg(process.pid, signal.SIGTERM)
                try:
                    process.wait(timeout=2)
                except subprocess.TimeoutExpired:
                    os.killpg(process.pid, signal.SIGKILL)
                    process.wait()
                break
            time.sleep(0.1)
    code = process.wait()
    result = dict(phase=phase, exit_code=code, resource_interruption=interruption,
                  termination_signal=-code if code < 0 else None,
                  elapsed_seconds=time.monotonic() - started, sampled_peak_group_rss_bytes=peak,
                  minimum_available_bytes=minimum, samples=samples,
                  maximum_observed_owned_processes=max_owned, policy=registration['resource_policy'])
    write(attempt / (phase + '-resource-outcome.json'), result)
    print(json.dumps(result), flush=True)
    print((attempt / (phase + '-stdout.log')).read_text()[-1200:], flush=True)
    print((attempt / (phase + '-stderr.log')).read_text()[-2500:], flush=True)
    if code != 0 or interruption:
        raise SystemExit(1)
    return result
