"""Curate exact bounded continuity records; never relabel the failed search qualification."""
import gzip
import hashlib
import json
from pathlib import Path

root = Path('/fast_data/home/liliana/repos@uibcdf/dockingmt')
task = root / '.molsyssuite/e2-source-admission-20261107'
out = root / 'devguide/validation/data/e2_source_admission_2026-11-07'
for label in ('source-01', 'reader-01'):
    assert json.loads((task / label / 'resource-outcome.json').read_text())['outcome'] == 'PASS'
    assert json.loads((task / label / 'output/receipt.json').read_text())['outcome'] == 'PASS'
out.mkdir(exist_ok=False)
files = []


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def keep(source, relative):
    raw = source.read_bytes()
    name = relative + ('.txt' if source.suffix in ('.py', '.ipynb') else '')
    compress = source.suffix in ('.json', '.ipynb', '.h5msm', '.log') or len(raw) > 50000
    if compress:
        name += '.gz'
    stored = gzip.compress(raw, mtime=0) if compress else raw
    destination = out / name
    destination.parent.mkdir(parents=True, exist_ok=True)
    with destination.open('xb') as stream:
        stream.write(stored)
    files.append(dict(path=name, original_path=relative, bytes=len(stored), sha256=sha(stored),
                      decoded_bytes=len(raw), decoded_sha256=sha(raw)))


for label in ('source-01', 'reader-01'):
    for path in sorted((task / label).rglob('*')):
        if path.is_file() and 'cache' not in path.relative_to(task / label).parts:
            keep(path, 'attempts/' + str(path.relative_to(task)))
for name in ('e2_source_admission_guard.py', 'e2_source_admission_worker.py', 'e2_source_admission_reader.py', 'seal_e2_source_admission.py', 'preserve_e2_source_admission.py'):
    keep(Path('/tmp') / name, 'harnesses/' + name)

                notebook_sha256=sha((root/'devguide/validation/e2_ligand_source_receiving.ipynb').read_bytes()),
                molecular_receiving_outcome='PASS', persistence_outcome='PASS',

                new_type_assignments=0, new_preparations=0, new_energies=0, new_searches=0,
